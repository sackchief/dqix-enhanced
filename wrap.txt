# Functions recompiled as <name>_orig, with the runtime's wrap_<name> around them
# (the game's own code still does all the work; the runtime only watches).
# <module> <address> <name>   (named, so the runtime's hook doesn't depend on
# the address: tools/region.py moves the addresses for the European ROM)

# Draws one glyph of a font into a tiled text canvas (4 bits per pixel):
# func_0205b36c(canvas, glyph, x, y, [sp] colour, [sp+4] font). Watched by
# text_hd.c, which notes where each letter went so the GPU renderer can draw
# it again in high resolution (Graphics > HD text).
main 0205b36c glyph_tiled

# Draws one glyph into a plain (not tiled) 4-bit buffer: func_02045864(x, y,
# glyph, buffer, [sp] width, [sp+4] height, [sp+8] colour, [sp+12] font). The
# battle messages over the field: each letter (with its shadow or outline) in
# a 16x16 texture, drawn by the 3D engine. Watched by text_hd.c, like above.
main 02045864 glyph_plain

# memcpy: the tags of letters follow them when the game copies them itself
# (the textures above go through a staging buffer).
main 02001a40 memcpy

# The field's frame (time of day, the map...): func_ov017_0218d0ec, once a
# frame while on the field. Watched by autosave.c (Modern > Save by itself
# every 10 minutes), which saves the adventure there, as the priest would.
ov017 0218d0ec field_frame
